function doPost(e) { try { const sheet = SpreadsheetApp .openById("1YNhYKkoV_JKjhREkmM7w-ntciX-_qTqaDI9xpB7PBgQ") .getSheetByName("Sheet1"); const data = JSON.parse(e.postData.contents); // Cek Jam Pelajaran const jamPelajaran = data.jamPelajaran || ""; // Aturan Jam ke-2: // Jam ke-2 baru boleh diisi mulai pukul 08.00 if (jamPelajaran === "Jam ke-2") { const sekarang = new Date(); const jam = sekarang.getHours(); if (jam < 8) { return ContentService .createTextOutput(JSON.stringify({ success: false, message: "Absensi Jam ke-2 baru dapat dilakukan mulai pukul 08.00" })) .setMimeType(ContentService.MimeType.JSON); } } // Simpan data ke Google Sheets sheet.appendRow([ data.tanggal, data.jam, data.guru, data.kelas, jamPelajaran, data.mapel, data.status ]); return ContentService .createTextOutput(JSON.stringify({ success: true, message: "Absensi berhasil disimpan" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, message: "Terjadi kesalahan: " + error.message })) .setMimeType(ContentService.MimeType.JSON); } } // Mengambil data absensi untuk ditampilkan di website function doGet(e) { try { const sheet = SpreadsheetApp .openById("1YNhYKkoV_JKjhREkmM7w-ntciX-_qTqaDI9xpB7PBgQ") .getSheetByName("Sheet1"); const data = sheet.getDataRange().getValues(); // Hanya kembalikan data jika ada isi if (data.length <= 1) { return ContentService .createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } // Buang header const rows = data.slice(1); const hasil = rows.map(function(row) { return { tanggal: row[0], jam: row[1], guru: row[2], kelas: row[3], jamPelajaran: row[4], mapel: row[5], status: row[6] }; }); return ContentService .createTextOutput(JSON.stringify(hasil)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, message: error.message })) .setMimeType(ContentService.MimeType.JSON); } }
